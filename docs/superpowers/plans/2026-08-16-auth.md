# Fetch'd Auth Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the client-generated `localStorage` identity with real Supabase Auth (username + password), scoping RLS to the authenticated user, without changing swipe/match/Gemini business logic beyond swapping the identity source.

**Architecture:** A new `src/lib/auth.ts` module wraps Supabase Auth (fake-email plumbing, session helpers). `App.tsx` gates all routes behind an auth check and adds `/signup`/`/login` routes. The existing `dogs.session_id`/`adopters.session_id` columns become `user_id uuid references auth.users(id)`, and RLS policies move from fully open to per-operation ownership checks. `swipes.swiper_id`/`target_id` are unaffected — they already store profile row ids, not identity ids.

**Tech Stack:** React 19, TypeScript, Vite, `@supabase/supabase-js` (already a dependency — no new packages), Vitest + React Testing Library, react-router-dom v7.

**Spec:** `docs/superpowers/specs/2026-08-16-auth-design.md` (and the source addendum it resolves, `docs/superpowers/authspec.md`)

## Global Constraints

- Do not change swipe/match/Gemini business logic beyond swapping the identity source (per authspec.md).
- No new dependencies — use the already-installed `@supabase/supabase-js`.
- Username normalization: lowercase, strip whitespace, append fixed domain `@fetchd.local`.
- Out of scope: password reset, real email verification, OAuth, migrating old localStorage-identified data.
- Old demo rows may be discarded/reseeded rather than migrated (per spec).
- Every task must pass `npm run test`, `npx tsc -b --noEmit`, and `npm run build` before committing — this is the project's established verification gate.

```ts
import { supabase } from '../supabaseClient';

const EMAIL_DOMAIN = '@fetchd.local';
const TAKEN_ERROR = 'That username is taken — try another.';
const LOGIN_ERROR = 'Incorrect username or password.';

export function normalizeUsername(raw: string): string {
  return raw.trim().toLowerCase().replace(/\s+/g, '');
}

function usernameToEmail(username: string): string {
  return `${normalizeUsername(username)}${EMAIL_DOMAIN}`;
}

export async function signUp(username: string, password: string): Promise<void> {
  const normalized = normalizeUsername(username);
  const { data, error } = await supabase.auth.signUp({
    email: usernameToEmail(normalized),
    password,
    options: { data: { username: normalized } },
  });
  if (error) {
    if (/already registered/i.test(error.message)) throw new Error(TAKEN_ERROR);
    throw error;
  }
  if (data.user && data.user.identities && data.user.identities.length === 0) {
    throw new Error(TAKEN_ERROR);
  }
}

export async function logIn(username: string, password: string): Promise<void> {
  const { error } = await supabase.auth.signInWithPassword({
    email: usernameToEmail(username),
    password,
  });
  if (error) throw new Error(LOGIN_ERROR);
}

export async function logOut(): Promise<void> {
  await supabase.auth.signOut();
}

export async function getCurrentUserId(): Promise<string | null> {
  const { data } = await supabase.auth.getSession();
  return data.session?.user.id ?? null;
}

export async function getCurrentUsername(): Promise<string | null> {
  const { data } = await supabase.auth.getSession();
  return (data.session?.user.user_metadata.username as string | undefined) ?? null;
}

export function onAuthStateChange(callback: (userId: string | null) => void): () => void {
  const { data } = supabase.auth.onAuthStateChange((_event, session) => {
    callback(session?.user.id ?? null);
  });
  return () => data.subscription.unsubscribe();
}
```

```ts
export interface OwnedProfile {
  type: ProfileType;
  id: string;
}

export async function fetchOwnedProfiles(userId: string): Promise<OwnedProfile[]> {
  const [dogsResult, adoptersResult] = await Promise.all([
    supabase.from('dogs').select('id').eq('user_id', userId),
    supabase.from('adopters').select('id').eq('user_id', userId),
  ]);
  if (dogsResult.error) throw dogsResult.error;
  if (adoptersResult.error) throw adoptersResult.error;
  const dogProfiles: OwnedProfile[] = (dogsResult.data ?? []).map((row: { id: string }) => ({
    type: 'dog',
    id: row.id,
  }));
  const adopterProfiles: OwnedProfile[] = (adoptersResult.data ?? []).map((row: { id: string }) => ({
    type: 'adopter',
    id: row.id,
  }));
  return [...dogProfiles, ...adopterProfiles];
}
```

```tsx
import { useState } from 'react';
import type { FormEvent } from 'react';
import { Link } from 'react-router-dom';
import { signUp } from '../lib/auth';
import { Logo } from '../components/Logo';

interface SignupProps {
  onSignedUp: () => void;
}

export function Signup({ onSignedUp }: SignupProps) {
  const [username, setUsername] = useState('');
  const [password, setPassword] = useState('');
  const [error, setError] = useState<string | null>(null);
  const [submitting, setSubmitting] = useState(false);

  async function handleSubmit(e: FormEvent) {
    e.preventDefault();
    setError(null);
    setSubmitting(true);
    try {
      await signUp(username, password);
      onSignedUp();
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Something went wrong.');
    } finally {
      setSubmitting(false);
    }
  }

  return (
    <div className="flex min-h-screen flex-col items-center justify-center px-6 py-10 text-center sm:px-10">
      <Logo variant="stacked" />
      <form onSubmit={handleSubmit} className="mt-8 flex w-full max-w-90 flex-col gap-4 text-left">
        <div>
          <label className="mb-1.5 block text-xs text-ink/70">Username</label>
          <input
            aria-label="username"
            required
            value={username}
            onChange={(e) => setUsername(e.target.value)}
            className="w-full min-h-10 rounded-full border border-divider bg-surface px-3.5 py-2 text-sm outline-none focus:border-clay"
          />
        </div>
        <div>
          <label className="mb-1.5 block text-xs text-ink/70">Password</label>
          <input
            aria-label="password"
            type="password"
            required
            value={password}
            onChange={(e) => setPassword(e.target.value)}
            className="w-full min-h-10 rounded-full border border-divider bg-surface px-3.5 py-2 text-sm outline-none focus:border-clay"
          />
        </div>
        {error && <p className="text-xs text-clay-700">{error}</p>}
        <button
          type="submit"
          disabled={submitting}
          className="rounded-full bg-clay px-7 py-3 font-heading text-[15px] text-cream transition-colors hover:bg-clay-600 disabled:opacity-60"
        >
          {submitting ? 'Signing up…' : 'Sign up'}
        </button>
      </form>
      <Link to="/login" className="mt-6 text-sm text-clay-700 underline underline-offset-4">
        Already have an account? Log in
      </Link>
    </div>
  );
}
```

- [ ] **Step 1: Write the migration file**

Create `supabase/migration-auth.sql`:

```sql
-- Replaces localStorage-generated session_id identity with real Supabase
-- Auth. Demo data seeded under the old identity scheme is discarded
-- (acceptable per docs/superpowers/specs/2026-08-16-auth-design.md).

truncate table messages, matches, swipes, dogs, adopters restart identity cascade;

alter table dogs drop column session_id;
alter table dogs add column user_id uuid not null references auth.users(id);

alter table adopters drop column session_id;
alter table adopters add column user_id uuid not null references auth.users(id);

drop policy "public read/write dogs" on dogs;
drop policy "public read/write adopters" on adopters;
drop policy "public read/write swipes" on swipes;
drop policy "public read/write matches" on matches;
drop policy "public read/write messages" on messages;
drop policy "public read/write photos bucket" on storage.objects;

-- dogs / adopters: any authenticated user can read (Discover shows other
-- people's profiles); only the owner can write.
create policy "authenticated read dogs" on dogs for select
  to authenticated using (true);
create policy "owner insert dogs" on dogs for insert
  to authenticated with check (user_id = auth.uid());
create policy "owner update dogs" on dogs for update
  to authenticated using (user_id = auth.uid()) with check (user_id = auth.uid());
create policy "owner delete dogs" on dogs for delete
  to authenticated using (user_id = auth.uid());

create policy "authenticated read adopters" on adopters for select
  to authenticated using (true);
create policy "owner insert adopters" on adopters for insert
  to authenticated with check (user_id = auth.uid());
create policy "owner update adopters" on adopters for update
  to authenticated using (user_id = auth.uid()) with check (user_id = auth.uid());
create policy "owner delete adopters" on adopters for delete
  to authenticated using (user_id = auth.uid());

-- swipes: swiper_id/target_id are dog/adopter row ids, not auth ids, so
-- ownership is checked via a join. Read stays open to authenticated users
-- since recordSwipe must read the *other* party's swipe row to detect a
-- mutual match.
create policy "authenticated read swipes" on swipes for select
  to authenticated using (true);
create policy "owner insert swipes" on swipes for insert
  to authenticated with check (
    exists (select 1 from dogs where dogs.id = swiper_id and dogs.user_id = auth.uid())
    or exists (select 1 from adopters where adopters.id = swiper_id and adopters.user_id = auth.uid())
  );

-- matches: only the two owning parties can see or update a match.
create policy "participant read matches" on matches for select
  to authenticated using (
    exists (select 1 from dogs where dogs.id = dog_id and dogs.user_id = auth.uid())
    or exists (select 1 from adopters where adopters.id = adopter_id and adopters.user_id = auth.uid())
  );
create policy "participant insert matches" on matches for insert
  to authenticated with check (
    exists (select 1 from dogs where dogs.id = dog_id and dogs.user_id = auth.uid())
    or exists (select 1 from adopters where adopters.id = adopter_id and adopters.user_id = auth.uid())
  );
create policy "participant update matches" on matches for update
  to authenticated using (
    exists (select 1 from dogs where dogs.id = dog_id and dogs.user_id = auth.uid())
    or exists (select 1 from adopters where adopters.id = adopter_id and adopters.user_id = auth.uid())
  ) with check (
    exists (select 1 from dogs where dogs.id = dog_id and dogs.user_id = auth.uid())
    or exists (select 1 from adopters where adopters.id = adopter_id and adopters.user_id = auth.uid())
  );

-- messages: only participants in the parent match can read/send.
create policy "participant read messages" on messages for select
  to authenticated using (
    exists (
      select 1 from matches
      join dogs on dogs.id = matches.dog_id
      join adopters on adopters.id = matches.adopter_id
      where matches.id = match_id
        and (dogs.user_id = auth.uid() or adopters.user_id = auth.uid())
    )
  );
create policy "participant insert messages" on messages for insert
  to authenticated with check (
    exists (
      select 1 from matches
      join dogs on dogs.id = matches.dog_id
      join adopters on adopters.id = matches.adopter_id
      where matches.id = match_id
        and (dogs.user_id = auth.uid() or adopters.user_id = auth.uid())
    )
  );

-- storage: photos stay publicly readable; only authenticated users may write.
create policy "public read photos bucket" on storage.objects for select
  using (bucket_id = 'photos');
create policy "authenticated insert photos bucket" on storage.objects for insert
  to authenticated with check (bucket_id = 'photos');
create policy "authenticated update photos bucket" on storage.objects for update
  to authenticated using (bucket_id = 'photos') with check (bucket_id = 'photos');
create policy "authenticated delete photos bucket" on storage.objects for delete
  to authenticated using (bucket_id = 'photos');
```
